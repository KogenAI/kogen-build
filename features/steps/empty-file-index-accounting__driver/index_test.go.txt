package build

import (
	"encoding/hex"
	"encoding/json"
	"os"
	"testing"
)

func TestEmptyFileIndexAcceptanceProbe(t *testing.T) {
	var input struct {
		Listing  string
		Required []string
	}
	if err := json.Unmarshal([]byte(os.Getenv("EMPTY_FILE_INDEX_INPUT")), &input); err != nil {
		t.Fatal(err)
	}
	listing, err := hex.DecodeString(input.Listing)
	if err != nil {
		t.Fatal(err)
	}
	output, err := json.Marshal([2]string{
		fileIndex(string(listing), nil),
		fileIndexBound(string(listing), input.Required, 0),
	})
	if err != nil {
		t.Fatal(err)
	}
	if err := os.WriteFile(os.Getenv("EMPTY_FILE_INDEX_OUTPUT"), output, 0600); err != nil {
		t.Fatal(err)
	}
}

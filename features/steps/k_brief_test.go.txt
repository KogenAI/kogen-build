package build

import (
 "fmt"
 "os"
 "os/exec"
 "path/filepath"
 "reflect"
 "strings"
 "testing"

 "github.com/KogenAI/kogen-coordinator/git"
 "github.com/KogenAI/kogen-coordinator/project"
)

func TestKBriefAcceptance(t *testing.T) {
 dir := physical(t, t.TempDir())
 env := isolatedEnv()
 cmd := exec.Command("git", "init", "-q"); cmd.Dir, cmd.Env = dir, env
 if out, err := cmd.CombinedOutput(); err != nil { t.Fatalf("git init: %v %s", err, out) }
 text := "# Brief fixture\nSlug: fixture\nModel: not applicable — fixture\nAcceptance:\n- .kogen/intents/fixture/acceptance.feature\n\n## Tests\n- none: fixture\n"
 feature := "Feature: authority\n  Scenario: complete\n    Then authority remains\n"
 model := "module fixture { val authority = true }\n"
 writeFile(t, filepath.Join(dir,".kogen/intents/fixture/intent.md"), text, 0644)
 writeFile(t, filepath.Join(dir,".kogen/intents/fixture/acceptance.feature"), feature, 0644)
 writeFile(t, filepath.Join(dir,".kogen/intents/fixture/model.qnt"), model, 0644)
 readme := strings.Repeat("R", 20480)
 writeFile(t, filepath.Join(dir,"README.md"), readme, 0644)
 for i:=0;i<496;i++ { writeFile(t, filepath.Join(dir,fmt.Sprintf("f%03d",i)), "x",0644) }
 in, err := project.LoadIntent(dir,"fixture"); if err != nil { t.Fatal(err) }
 b := &Attempt{State:State{Root:dir,Slug:"fixture",Config:&project.Config{Setup:[]string{"setup-authority"},Fix:[]string{"fix-authority"},Checks:[]project.Check{{Name:"authority",Run:"check-authority"}},Acceptance:&project.Acceptance{Run:"acceptance-authority"}},Baseline:[]string{"protected-authority"}},intent:in,repo:git.New(dir,env)}
 original,err := b.brief(); if err != nil { t.Fatal(err) }
 mode := os.Getenv("KB_MODE")
 if mode == "full" { return }
 if mode != "good" && mode != "bad" {
  for name,value := range map[string]int{"BuildBriefMaxFiles":200,"BuildBriefMaxReadmeBytes":8192} {
   f:=reflect.ValueOf(&b.deps).Elem().FieldByName(name)
   if !f.IsValid() || !f.CanSet() || f.Kind()!=reflect.Int { t.Fatalf("missing internal Deps.%s int",name) }; f.SetInt(int64(value))
  }
 }
 got,err := b.brief(); if err != nil { t.Fatal(err) }
 noteR := "[README.md cut: 12288 bytes omitted; read_file README.md for the rest.]"
 noteF := "[File list cut: 300 entries omitted; use shell git ls-files --cached --others --exclude-standard for the rest.]"
 if mode == "good" || mode == "bad" {
  got.Stable = strings.Replace(original.Stable,readme,readme[:8192]+"\n"+noteR,1)
  start:=strings.Index(original.Dynamic,"# Files\n\n")+len("# Files\n\n")
  lines:=strings.Split(strings.TrimSuffix(original.Dynamic[start:],"\n"),"\n")
  if len(lines)!=500 { t.Fatalf("fixture has %d files",len(lines)) }
  got.Dynamic=original.Dynamic[:start]+strings.Join(lines[:200],"\n")+"\n"+noteF
  if mode=="bad" { got.Dynamic += "\nextra-file" }
 }
 wantStable:=strings.Replace(original.Stable,readme,readme[:8192]+"\n"+noteR,1)
 if got.Stable!=wantStable { t.Fatal("capped stable block differs: README prefix, note or authority") }
 start:=strings.Index(original.Dynamic,"# Files\n\n")+len("# Files\n\n")
 lines:=strings.Split(strings.TrimSuffix(original.Dynamic[start:],"\n"),"\n")
 if len(lines)!=500 { t.Fatalf("fixture has %d files",len(lines)) }
 wantDynamic:=original.Dynamic[:start]+strings.Join(lines[:200],"\n")+"\n"+noteF
 if got.Dynamic!=wantDynamic { t.Fatal("capped dynamic block differs: prefix, file entries or note") }
}

const unmovedCause = "observed output was never produced"
func unmovedPrefix(n int, common bool) string {
 s := fmt.Sprintf("Failing set did not move for %d consecutive red passes. Last round changed: unrelated.txt. None of the failing items changed.", n)
 if common { s += "\nCommon failure text (address first): " + unmovedCause }
 return s + "\n\n"
}
func unmovedOrdinary(s string) string {
 i := strings.Index(s, "Verification pass ")
 if i < 0 { return s }; return s[i:]
}
func unmovedValidate(feedback []string, rounds int, common bool) error {
 if len(feedback) != rounds { return fmt.Errorf("received %d red feedback messages, want %d",len(feedback),rounds) }
 first := unmovedOrdinary(feedback[0])
 if feedback[0] != first { return fmt.Errorf("first red changed: %s",feedback[0]) }
 for i:=1;i<rounds;i++ {
  prefix:=unmovedPrefix(i+1,common)
  if !strings.HasPrefix(feedback[i],prefix) { return fmt.Errorf("pass %d missing prefix %q: %s",i+1,prefix,feedback[i]) }
  want:=strings.Replace(first,"Verification pass 1 ",fmt.Sprintf("Verification pass %d ",i+1),1)
  if feedback[i][len(prefix):] != want { return fmt.Errorf("pass %d reordered or changed ordinary feedback\ngot %s\nwant %s",i+1,feedback[i][len(prefix):],want) }
 }
 return nil
}
func TestKUnmovedAcceptance(t *testing.T) {
 mode:=os.Getenv("KU_MODE")
 rounds:=3
 if mode=="history" { rounds=9 }
 e:=newEnv(t)
 feature:="Feature: Unproduced output\n"
 for i:=1;i<=12;i++ { feature+=fmt.Sprintf("  Scenario: Item %d\n    Then output %d exists\n",i,i) }
 e.write(project.FeaturePath(testSlug),feature)
 scenarios,err:=project.ParseGherkin(feature);if err!=nil {t.Fatal(err)}
 report:=func(partial, differing bool) string {
  events:=[]obj{documentMessage(project.FeaturePath(testSlug),scenarios)}
  for i,s:=range scenarios {
   result:=obj{"status":"FAILED","message":unmovedCause}
   if partial && i==0 {result=obj{"status":"PASSED"}}
   if differing {result["message"]=fmt.Sprintf("different failure %d",i+1)}
   events=append(events,caseMessages(s,i+1,project.FeaturePath(testSlug),[]obj{result})...)
  }
  var lines []string
  for _,event:=range events {b,err:=core.EncodeJSON(event);if err!=nil {t.Fatal(err)};lines=append(lines,string(b))}
  return strings.Join(lines,"\n")+"\n"
 }
 different:=mode=="different"
 writeFile(t,filepath.Join(e.aux,"red.ndjson"),report(false,different),0644)
 writeFile(t,filepath.Join(e.aux,"partial.ndjson"),report(true,false),0644)
 writeFile(t,filepath.Join(e.aux,"accept.sh"),"#!/bin/sh\nif test -f partial.txt; then cp '"+filepath.Join(e.aux,"partial.ndjson")+"' \"$KOGEN_ACCEPTANCE_REPORT\"; else cp '"+filepath.Join(e.aux,"red.ndjson")+"' \"$KOGEN_ACCEPTANCE_REPORT\"; fi\nexit 1\n",0755)
 e.approve()
 for i:=1;i<=rounds;i++ {
  file:="unrelated.txt";if mode=="partial" && i==2 {file="partial.txt"}
  e.prov.then(e.prov.calls("write_file",map[string]any{"path":file,"content":fmt.Sprintf("round %d\n",i)},"finish",map[string]any{"summary":"another round"}))
 }
 e.prov.then(sayings(3)...)
 a:=e.build()
 if a.exit!=1 || a.err!=nil {t.Fatalf("exit %d err %v: %s",a.exit,a.err,a.shown())}
 if got:=e.result(e.onlyRun())["outcome"];got!="no_progress" {t.Fatalf("unexpected stop: %v",got)}
 reqs:=e.prov.requests()
 if len(reqs)!=rounds+2 {t.Fatalf("Build stopped while edits continued: requests %d want %d",len(reqs),rounds+2)}
 var feedback []string
 for i:=1;i<=rounds;i++ {feedback=append(feedback,lastInput(reqs[i]))}
 notice:=lastInput(reqs[rounds+1])
 if !strings.HasPrefix(notice,"Required acceptance items are still missing:") || strings.Contains(notice,"Failing set did not move") {t.Fatalf("tool-free notice replaced: %s",notice)}
 if mode=="partial" {
  if strings.Contains(feedback[1],"Failing set did not move") {t.Fatalf("changed scenario set wrongly reported unchanged: %s",feedback[1])}
  if !strings.Contains(feedback[1],"Item 2") {t.Fatal("partial feedback did not reach remaining failures")}
  return
 }
 if mode=="control-good" || mode=="control-bad" {
  for i:=range feedback {feedback[i]=unmovedOrdinary(feedback[i]);if i>0 {feedback[i]=unmovedPrefix(i+1,true)+feedback[i]}}
  if mode=="control-bad" {feedback[2]=strings.Replace(feedback[2],"consecutive red passes","consecutive green passes",1)}
 }
 if err:=unmovedValidate(feedback,rounds,!different);err!=nil {t.Fatal(err)}
}

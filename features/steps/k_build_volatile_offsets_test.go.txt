package build

// The acceptance harness of k-build-volatile-offsets. Its acceptance step overlays it into ./build
// as an internal test file and runs TestKBuildVolatileOffsetsAcceptance once. The test runs one
// whole Build against the package's scripted provider (fakeProvider), then rewrites one value of
// the first provider attempt record (the "provider" object of the first provider_attempt event) in
// the run's log.jsonl at a time and compares observeBuild of the rewritten run with observeBuild of
// the recorded one (new keys: two rewritten runs with each other). It prints one KBVO line per
// variant and judges nothing itself: the step compares the lines with the feature's expected values.

import (
	"encoding/json"
	"fmt"
	"os"
	"strings"
	"testing"
)

// kbvoDurations adds 4321 to every number under a key ending in _ms, at any depth.
func kbvoDurations(v any) any {
	switch x := v.(type) {
	case map[string]any:
		for k, val := range x {
			if n, ok := val.(float64); ok && strings.HasSuffix(k, "_ms") {
				x[k] = n + 4321
			} else {
				x[k] = kbvoDurations(val)
			}
		}
	case []any:
		for i, val := range x {
			x[i] = kbvoDurations(val)
		}
	}
	return v
}

func TestKBuildVolatileOffsetsAcceptance(t *testing.T) {
	e := newEnv(t)
	f := e.prov
	f.then(landScript(f)...)
	if a := e.build(); a.exit != 0 {
		t.Fatalf("exit %d: %s", a.exit, a.shown())
	}
	id := e.onlyRun()
	logPath := e.runPath(id, "log.jsonl")
	raw, err := os.ReadFile(logPath)
	if err != nil {
		t.Fatal(err)
	}
	original := string(raw)
	lines := strings.Split(strings.TrimSuffix(original, "\n"), "\n")
	at := -1
	var record map[string]any
	for i, line := range lines {
		var doc map[string]any
		if err := json.Unmarshal([]byte(line), &doc); err != nil {
			t.Fatal(err)
		}
		if p, ok := doc["provider"].(map[string]any); ok && doc["event"] == "provider_attempt" {
			at, record = i, p
			break
		}
	}
	if at < 0 {
		t.Fatalf("no provider_attempt event in %s:\n%s", logPath, original)
	}
	_, offset := record["request_offset_ms"]
	_, load := record["load_average"]
	fmt.Printf("KBVO attempt record request_offset_ms=%v load_average=%v\n", offset, load)
	// observe is observeBuild of the run with the attempt record rewritten by change (the event's
	// own "at" by changeEvent).
	observeEvent := func(change, changeEvent func(map[string]any)) budgetObservation {
		var doc map[string]any
		if err := json.Unmarshal([]byte(lines[at]), &doc); err != nil {
			t.Fatal(err)
		}
		change(doc["provider"].(map[string]any))
		changeEvent(doc)
		b, err := json.Marshal(doc)
		if err != nil {
			t.Fatal(err)
		}
		rewritten := append([]string{}, lines...)
		rewritten[at] = string(b)
		if err := os.WriteFile(logPath, []byte(strings.Join(rewritten, "\n")+"\n"), 0o600); err != nil {
			t.Fatal(err)
		}
		defer os.WriteFile(logPath, raw, 0o600)
		return observeBuild(t, e, id)
	}
	observe := func(change func(map[string]any)) budgetObservation {
		return observeEvent(change, func(map[string]any) {})
	}
	compare := func(name string, a, b budgetObservation) {
		var differs []string
		for _, p := range [][3]string{
			{"requests", a.requests, b.requests}, {"log", a.log, b.log}, {"checkpoint", a.checkpoint, b.checkpoint},
			{"result", a.result, b.result}, {"commit", a.outcome, b.outcome},
		} {
			if p[1] != p[2] {
				differs = append(differs, p[0])
			}
		}
		if len(differs) == 0 {
			fmt.Printf("KBVO %s equal\n", name)
		} else {
			fmt.Printf("KBVO %s differs: %s\n", name, strings.Join(differs, ","))
		}
	}
	base := observe(func(map[string]any) {})
	set := func(pairs ...any) func(map[string]any) {
		return func(doc map[string]any) {
			for i := 0; i < len(pairs); i += 2 {
				doc[pairs[i].(string)] = pairs[i+1]
			}
		}
	}
	for _, v := range []struct {
		name   string
		change func(map[string]any)
	}{
		{"request_offset_ms", set("request_offset_ms", 987654)},
		{"load_average", set("load_average", []any{99.99, 88.88, 77.77})},
		{"load_average null", set("load_average", nil)},
		{"every _ms number", func(doc map[string]any) { kbvoDurations(doc) }},
		{"timestamp", set("timestamp", "2030-01-02T03:04:05.678901Z")},
		{"provider_request_id", set("provider_request_id", "req-changed")},
		{"pass", set("pass", 7)},
		{"outcome", set("outcome", "failed")},
		{"stream_bytes", func(doc map[string]any) { doc["stream_bytes"] = doc["stream_bytes"].(float64) + 1 }},
	} {
		compare(v.name, base, observe(v.change))
	}
	compare("event at", base, observeEvent(func(map[string]any) {}, set("at", "2030-01-02T03:04:05.678901Z")))
	// Keys the record does not carry today: two runs that differ only in their values.
	compare("new timing keys",
		observe(set("spawn_offset_ms", 1, "queue_ms", 2, "ended_at", "2030-01-02T03:04:05Z")),
		observe(set("spawn_offset_ms", 801, "queue_ms", 902, "ended_at", "2031-06-07T08:09:10Z")))
	compare("new non-timing key",
		observe(set("attempts_seen", 1)),
		observe(set("attempts_seen", 2)))
}

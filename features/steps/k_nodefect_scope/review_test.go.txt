package build

import (
	"encoding/json"
	"fmt"
	"os"
	"strings"
	"testing"
)

// The k-nodefect-scope acceptance harness, compiled into package build through `go test
// -overlay` by harness.js; it is not a file of the package. It runs one real Build of the
// newEnv fixture project with review_material mode KNODEFECT_SCOPE_MODE: the scripted builder
// changes app.txt, and every scripted turn of the real reviewer submits a changes verdict whose
// only finding is the blocking KNODEFECT_SCOPE_MESSAGE at app.txt:1 (four turns: a review, its
// rejected-verdict allowance, the automatic fresh review and its allowance). It records what the
// Build did in the JSON file KNODEFECT_SCOPE_OUT and fails only when the harness cannot run.
func TestKNodefectScopeAcceptance(t *testing.T) {
	mode, message, out := os.Getenv("KNODEFECT_SCOPE_MODE"), os.Getenv("KNODEFECT_SCOPE_MESSAGE"), os.Getenv("KNODEFECT_SCOPE_OUT")
	if (mode != "on_demand" && mode != "parts") || message == "" || out == "" {
		t.Fatalf("KNODEFECT_SCOPE_MODE (on_demand or parts), KNODEFECT_SCOPE_MESSAGE and KNODEFECT_SCOPE_OUT are required; mode was %q", mode)
	}
	e := newEnv(t)
	e.write(".kogen/project.yaml", e.read(".kogen/project.yaml")+"review_material:\n  mode: "+mode+"\n")
	e.run("git", "add", ".kogen/project.yaml")
	e.run("git", "commit", "-qm", "review material mode")
	e.approve()
	f := e.prov
	f.then(f.calls("write_file", map[string]any{"path": "app.txt", "content": "candidate change\n"}, "finish", map[string]any{"summary": "done"}))
	for i := 0; i < 4; i++ {
		f.then(submit(f, "changes", finding("blocking", message)))
	}
	start := e.head()
	a := e.reviewed()
	run := e.onlyRun()
	verdicts := []string{}
	problems := []string{}
	for _, ev := range eventsNamed(e.events(run), "review") {
		verdicts = append(verdicts, fmt.Sprint(ev["verdict"]))
		if list, ok := ev["problems"].([]any); ok {
			for _, p := range list {
				problems = append(problems, fmt.Sprint(p))
			}
		}
	}
	record := map[string]any{
		"mode":             mode,
		"exit":             a.exit,
		"outcome":          e.result(run)["outcome"],
		"head_unchanged":   e.head() == start,
		"review_verdicts":  verdicts,
		"invalid_problems": problems,
		"feedback":         strings.Join(f.toolOutputs(), "\n"),
		"shown":            a.shown(),
	}
	raw, err := json.MarshalIndent(record, "", "  ")
	if err != nil {
		t.Fatal(err)
	}
	if err := os.WriteFile(out, append(raw, '\n'), 0o600); err != nil {
		t.Fatal(err)
	}
}
